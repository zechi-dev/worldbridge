create or replace function attach_referral(p_worker_tg_id bigint, p_code text) returns text language plpgsql security definer set search_path = public as $$ declare s bigint; begin select telegram_id into s from sourcers where upper(referral_code) = upper(p_code) limit 1; if s is null then return 'bad_code'; end if; if s = p_worker_tg_id then return 'self'; end if; update gulf_leads set referred_by_sourcer_tg_id = s::text where telegram_id::text = p_worker_tg_id::text and (referred_by_sourcer_tg_id is null or referred_by_sourcer_tg_id::text = ''); if found then return 'attached'; end if; if exists (select 1 from gulf_leads where telegram_id::text = p_worker_tg_id::text) then return 'already'; end if; return 'no_worker'; end $$; grant execute on function attach_referral(bigint, text) to anon, authenticated;
